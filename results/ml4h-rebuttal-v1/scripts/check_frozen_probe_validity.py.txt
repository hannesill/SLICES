#!/usr/bin/env python3
"""Validity checks for the ML4H rebuttal converged frozen-probe diagnostic.

Establishes that the cached-embedding probe path matches the frozen benchmark
linear-probe path before any converged-probe test metric is trusted:

A. ``encoders``   - local ``encoder.pt`` SHA256 vs the frozen encoder audit.
B. ``splits``     - train/test row counts, positives, resolved class weights and
                    LOS constant baselines vs the frozen thesis export.
C. ``embeddings`` - cached embeddings vs the benchmark ``FineTuneModule``
                    forward pass on ``ICUDataModule`` batches (and stay/label
                    parity of the task split).
D. ``capped``     - re-run of the benchmark's capped AdamW probe recipe on the
                    cached embeddings vs frozen same-seed ``linear_probe`` rows.

Read-only with respect to ``results/slices-thesis-v1``; writes JSON/CSV under
``outputs/ml4h-rebuttal/validity``.
"""

def parse_args(argv: Sequence[str] | None = None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--encoder-archive", type=Path, required=True)
    parser.add_argument("--processed-root", type=Path, required=True)
    parser.add_argument(
        "--embedding-root", type=Path, default=Path("outputs/ml4h-rebuttal/embeddings")
    )
    parser.add_argument("--output-dir", type=Path, default=DEFAULT_OUTPUT)
    parser.add_argument(
        "--checks",
        nargs="+",
        default=["encoders", "splits", "embeddings", "capped"],
        choices=("encoders", "splits", "embeddings", "capped", "capped_seed_sensitivity"),
    )
    parser.add_argument(
        "--capped-cells",
        nargs="*",
        default=None,
        help="Override capped-probe cells as paradigm:dataset:seed:task.",
    )
    return parser.parse_args(argv)


def _write(output_dir: Path, name: str, payload: Any) -> None:
    output_dir.mkdir(parents=True, exist_ok=True)
    if isinstance(payload, pd.DataFrame):
        payload.to_csv(output_dir / f"{name}.csv", index=False)
    else:
        (output_dir / f"{name}.json").write_text(
            json.dumps(payload, sort_keys=True, indent=1) + "\n"
        )

# A. Encoder provenance
# ---------------------------------------------------------------------------


def check_encoders(args: argparse.Namespace) -> dict[str, Any]:
    audit = pd.read_csv(AUDIT_CSV)
    rows = []
    for record in audit.itertuples():
        path = encoder_path(
            args.encoder_archive, record.paradigm, record.dataset, int(record.seed), "encoder.pt"
        )
        local = file_sha256(path) if path.exists() else None
        rows.append(
            {
                "run": record.run,
                "local_sha256": local,
                "audit_final_encoder_sha256": record.final_encoder_sha256,
                "match": local == record.final_encoder_sha256,
            }
        )
    frame = pd.DataFrame(rows)
    _write(REPO_ROOT / args.output_dir, "encoder_sha256_check", frame)
    summary = {"n_encoders": int(len(frame)), "n_match": int(frame["match"].sum())}
    print(f"[encoders] {summary}", flush=True)
    return summary

# B. Split / label parity against the frozen export
# ---------------------------------------------------------------------------


def check_splits(args: argparse.Namespace) -> dict[str, Any]:
    per_seed = pd.read_parquet(FROZEN_DIR / "per_seed_results.parquet")
    long = pd.read_parquet(FROZEN_DIR / "metrics_long.parquet")
    configs = pd.read_parquet(FROZEN_DIR / "run_configs.parquet")
    probe = per_seed[
        (per_seed.experiment_class == "core_ssl_benchmark")
        & (per_seed.protocol == "linear_probe")
        & (per_seed.label_fraction == 1.0)
    ]
    sup_long = long[
        (long.experiment_class == "core_ssl_benchmark") & (long.paradigm == "supervised")
    ]

    def frozen_scalar(dataset: str, task: str, seed: int, metric: str) -> float | None:
        values = sup_long[
            (sup_long.dataset == dataset)
            & (sup_long.task == task)
            & (sup_long.seed == seed)
            & (sup_long.metric_name == metric)
        ].metric_value
        return float(values.iloc[0]) if len(values) else None

    rows = []
    for dataset in DEFAULT_DATASETS:
        for seed in DEFAULT_SEEDS:
            for task in DEFAULT_TASKS:
                split = load_task_split(args.processed_root / dataset, task, seed)
                y_train, y_test = split.labels["train"], split.labels["test"]
                row: dict[str, Any] = {
                    "dataset": dataset,
                    "seed": seed,
                    "task": task,
                    "n_train": len(y_train),
                    "n_test": len(y_test),
                    "frozen_n_test": frozen_scalar(dataset, task, seed, "test/n_samples"),
                }
                frozen_probe = probe[
                    (probe.dataset == dataset) & (probe.task == task) & (probe.seed == seed)
                ]
                if task in CLASSIFICATION_TASKS:
                    weights = sqrt_balanced_class_weights(y_train.astype(np.int64), task)
                    cfg_weights = set()
                    for run_id in frozen_probe.wandb_run_id:
                        cfg = json.loads(
                            configs.loc[configs.wandb_run_id == run_id, "config_json"].iloc[0]
                        )
                        cfg_weights.add(tuple(cfg["training"]["class_weight"]))
                    frozen_w = sorted(cfg_weights)[0] if len(cfg_weights) == 1 else None
                    row.update(
                        {
                            "n_train_positive": int(y_train.sum()),
                            "frozen_n_train": float(
                                frozen_probe["train_label_support/full_train_total"].iloc[0]
                            ),
                            "frozen_n_train_positive": float(
                                frozen_probe["train_label_support/full_train_positive"].iloc[0]
                            ),
                            "n_test_positive": int(y_test.sum()),
                            "frozen_n_test_positive": frozen_scalar(
                                dataset, task, seed, "test/n_positive"
                            ),
                            "class_weight_max_abs_diff": (
                                float(np.max(np.abs(np.asarray(weights) - np.asarray(frozen_w))))
                                if frozen_w is not None
                                else None
                            ),
                        }
                    )
                else:
                    y_tr = y_train.astype(np.float64)
                    y_te = y_test.astype(np.float64)
                    mean_mse = regression_metrics(y_te, np.full_like(y_te, y_tr.mean()))["mse"]
                    median_mae = regression_metrics(y_te, np.full_like(y_te, np.median(y_tr)))[
                        "mae"
                    ]
                    row.update(
                        {
                            "train_label_mean": float(y_tr.mean()),
                            "frozen_train_label_mean": frozen_scalar(
                                dataset, task, seed, "baseline/train_label_mean"
                            ),
                            "train_label_median": float(np.median(y_tr)),
                            "frozen_train_label_median": frozen_scalar(
                                dataset, task, seed, "baseline/train_label_median"
                            ),
                            "mean_predictor_test_mse": mean_mse,
                            "frozen_mean_predictor_mse": frozen_scalar(
                                dataset, task, seed, "baseline/mean_predictor_mse"
                            ),
                            "median_predictor_test_mae": median_mae,
                            "frozen_median_predictor_mae": frozen_scalar(
                                dataset, task, seed, "baseline/median_predictor_mae"
                            ),
                        }
                    )
                rows.append(row)
    frame = pd.DataFrame(rows)
    _write(REPO_ROOT / args.output_dir, "split_label_parity", frame)

    cls = frame[frame.task.isin(CLASSIFICATION_TASKS)]
    reg = frame[~frame.task.isin(CLASSIFICATION_TASKS)]
    summary = {
        "n_cells": int(len(frame)),
        "n_train_match": int((cls.n_train == cls.frozen_n_train).sum()),
        "n_train_positive_match": int((cls.n_train_positive == cls.frozen_n_train_positive).sum()),
        "n_classification_cells": int(len(cls)),
        "n_test_match": int((frame.n_test == frame.frozen_n_test).sum()),
        "n_test_positive_match": int((cls.n_test_positive == cls.frozen_n_test_positive).sum()),
        "class_weight_max_abs_diff": float(cls.class_weight_max_abs_diff.max()),
        "n_regression_cells": int(len(reg)),
        "los_train_mean_max_abs_diff": float(
            (reg.train_label_mean - reg.frozen_train_label_mean).abs().max()
        ),
        "los_train_median_max_abs_diff": float(
            (reg.train_label_median - reg.frozen_train_label_median).abs().max()
        ),
        "los_mean_predictor_mse_max_abs_diff": float(
            (reg.mean_predictor_test_mse - reg.frozen_mean_predictor_mse).abs().max()
        ),
        "los_median_predictor_mae_max_abs_diff": float(
            (reg.median_predictor_test_mae - reg.frozen_median_predictor_mae).abs().max()
        ),
    }
    print(f"[splits] {summary}", flush=True)
    return summary

@torch.inference_mode()
def check_embeddings(args: argparse.Namespace) -> dict[str, Any]:
    rows = []
    for paradigm, dataset, seed, task in DEFAULT_EMBEDDING_CELLS:
        processed_dir = (args.processed_root / dataset).resolve()
        ckpt = encoder_path(args.encoder_archive, paradigm, dataset, seed, "encoder.pt")
        cfg = compose_probe_cfg(
            dataset=dataset, seed=seed, task=task, processed_dir=processed_dir, checkpoint=ckpt
        )
        pl.seed_everything(seed, workers=True)
        datamodule = ICUDataModule(
            processed_dir=processed_dir,
            task_name=task,
            batch_size=cfg.training.batch_size,
            num_workers=0,
            seed=seed,
            label_fraction=1.0,
        )
        datamodule.setup()
        model = build_frozen_probe_module(
            cfg,
            feature_dim=datamodule.get_feature_dim(),
            seq_length=datamodule.get_seq_length(),
            checkpoint=ckpt,
        )
        captured: list[torch.Tensor] = []
        handle = model.task_head.register_forward_pre_hook(
            lambda _module, inputs: captured.append(inputs[0].detach().cpu())
        )
        embeddings, emb_stay_ids, _meta = load_embedding_file(
            REPO_ROOT / args.embedding_root / dataset / f"seed{seed}" / f"{paradigm}.npz"
        )
        row_by_stay = {int(s): i for i, s in enumerate(emb_stay_ids)}
        split = load_task_split(processed_dir, task, seed)

        for split_name, loader, max_batches in (
            ("test", datamodule.test_dataloader(), None),
            ("val", datamodule.val_dataloader(), 20),
        ):
            captured.clear()
            stay_ids: list[torch.Tensor] = []
            labels: list[torch.Tensor] = []
            for batch_idx, batch in enumerate(loader):
                if max_batches is not None and batch_idx >= max_batches:
                    break
                model(batch["timeseries"], batch["mask"])
                stay_ids.append(batch["stay_id"])
                labels.append(batch["label"])
            module_emb = torch.cat(captured).numpy()
            ids = torch.cat(stay_ids).numpy()
            lab = torch.cat(labels).numpy()
            cached = embeddings[rows_for(ids, row_by_stay)]
            n = len(ids)
            rows.append(
                {
                    "paradigm": paradigm,
                    "dataset": dataset,
                    "seed": seed,
                    "task": task,
                    "split": split_name,
                    "n_rows_compared": int(n),
                    "max_abs_diff": float(np.max(np.abs(module_emb - cached))),
                    "embedding_abs_mean": float(np.mean(np.abs(module_emb))),
                    "stay_id_order_match": bool(
                        np.array_equal(ids, split.stay_ids[split_name][:n])
                    ),
                    "label_match": bool(np.array_equal(lab, split.labels[split_name][:n])),
                    "split_size_match": bool(
                        len(getattr(datamodule, f"{split_name}_indices"))
                        == len(split.stay_ids[split_name])
                    ),
                }
            )
        train_ids = np.asarray(
            [datamodule.dataset.stay_ids[i] for i in datamodule.train_indices], dtype=np.int64
        )
        rows.append(
            {
                "paradigm": paradigm,
                "dataset": dataset,
                "seed": seed,
                "task": task,
                "split": "train",
                "n_rows_compared": int(len(train_ids)),
                "stay_id_order_match": bool(np.array_equal(train_ids, split.stay_ids["train"])),
                "label_match": bool(
                    np.array_equal(
                        datamodule.dataset._labels_tensor[datamodule.train_indices].numpy(),
                        split.labels["train"],
                    )
                ),
                "split_size_match": True,
            }
        )
        handle.remove()
    frame = pd.DataFrame(rows)
    _write(REPO_ROOT / args.output_dir, "embedding_parity", frame)
    summary = {
        "n_comparisons": int(len(frame)),
        "max_abs_diff": float(frame.max_abs_diff.max()),
        "all_stay_id_order_match": bool(frame.stay_id_order_match.all()),
        "all_label_match": bool(frame.label_match.all()),
        "all_split_size_match": bool(frame.split_size_match.all()),
    }
    print(f"[embeddings] {summary}", flush=True)
    return summary


# ---------------------------------------------------------------------------
# D. Capped benchmark probe recipe on cached embeddings
# ---------------------------------------------------------------------------


def run_capped_probe(
    args: argparse.Namespace,
    paradigm: str,
    dataset: str,
    seed: int,
    task: str,
    probe_seed: int | None = None,
) -> dict[str, Any]:
    """Train the benchmark's capped probe recipe on cached embeddings.

    ``probe_seed`` only changes the head initialisation and batch order; the
    encoder, split and data stay tied to ``seed``.
    """
    processed_dir = (args.processed_root / dataset).resolve()
    ckpt = encoder_path(args.encoder_archive, paradigm, dataset, seed, "encoder.pt")
    run_dir = (
        REPO_ROOT / args.output_dir / "capped_runs" / dataset / task / f"seed{seed}" / paradigm
    )
    if probe_seed is not None:
        run_dir = run_dir / f"probe_seed{probe_seed}"
    cfg = compose_probe_cfg(
        dataset=dataset,
        seed=seed,
        task=task,
        processed_dir=processed_dir,
        checkpoint=ckpt,
        extra_overrides=("training.accelerator=cpu", "training.devices=1"),
    )
    split = load_task_split(processed_dir, task, seed)
    embeddings, emb_stay_ids, _meta = load_embedding_file(
        REPO_ROOT / args.embedding_root / dataset / f"seed{seed}" / f"{paradigm}.npz"
    )
    row_by_stay = {int(s): i for i, s in enumerate(emb_stay_ids)}
    tensors = {
        name: (
            torch.from_numpy(embeddings[rows_for(ids, row_by_stay)]).float(),
            torch.from_numpy(split.labels[name]),
        )
        for name, ids in split.stay_ids.items()
    }

    class_weight = None
    OmegaConf.set_struct(cfg, False)
    if task in CLASSIFICATION_TASKS:
        weights = sqrt_balanced_class_weights(split.labels["train"].astype(np.int64), task)
        cfg.training.class_weight = list(weights)
        class_weight = torch.tensor(list(weights), dtype=torch.float32)
    else:
        cfg.training.class_weight = None
    OmegaConf.set_struct(cfg, True)

    def loader(name: str, shuffle: bool) -> DataLoader:
        return DataLoader(
            TensorDataset(*tensors[name]),
            batch_size=cfg.training.batch_size,
            shuffle=shuffle,
            drop_last=shuffle,
        )

    monitor, mode = _monitor_and_mode(cfg)
    checkpoint_callback = ModelCheckpoint(
        dirpath=run_dir / "checkpoints",
        filename="capped-probe-{epoch:03d}",
        monitor=monitor,
        mode=mode,
        save_top_k=1,
        auto_insert_metric_name=False,
    )
    early_stopping = EarlyStopping(
        monitor=monitor, patience=cfg.training.early_stopping_patience, mode=mode
    )
    pl.seed_everything(cfg.seed if probe_seed is None else probe_seed, workers=True)
    module = CachedLinearProbeModule(
        cfg=cfg, input_dim=int(tensors["train"][0].shape[1]), class_weight=class_weight
    )
    trainer = pl.Trainer(
        max_epochs=cfg.training.max_epochs,
        accelerator="cpu",
        devices=1,
        precision=cfg.training.get("precision", 32),
        callbacks=[checkpoint_callback, early_stopping],
        logger=False,
        gradient_clip_val=cfg.training.get("gradient_clip_val", None),
        enable_progress_bar=False,
        enable_model_summary=False,
    )
    trainer.fit(module, loader("train", True), loader("val", False))
    stopped_epoch = int(trainer.current_epoch)
    best = torch.load(checkpoint_callback.best_model_path, map_location="cpu", weights_only=False)
    module.load_state_dict(best["state_dict"])
    metrics = trainer.test(module, loader("test", False), verbose=False)[0]
    return {
        "paradigm": paradigm,
        "dataset": dataset,
        "seed": seed,
        "task": task,
        "epochs_run": stopped_epoch,
        "best_epoch": int(best["epoch"]),
        **{key: float(value) for key, value in metrics.items()},
    }


def check_capped(args: argparse.Namespace) -> dict[str, Any]:
    per_seed = pd.read_parquet(FROZEN_DIR / "per_seed_results.parquet")
    probe = per_seed[
        (per_seed.experiment_class == "core_ssl_benchmark")
        & (per_seed.protocol == "linear_probe")
        & (per_seed.label_fraction == 1.0)
    ]
    cells = DEFAULT_CAPPED_CELLS
    if args.capped_cells:
        cells = tuple(
            (p, d, int(s), t) for p, d, s, t in (cell.split(":") for cell in args.capped_cells)
        )
    rows = []
    out_dir = REPO_ROOT / args.output_dir
    for paradigm, dataset, seed, task in cells:
        result = run_capped_probe(args, paradigm, dataset, seed, task)
        frozen = probe[
            (probe.paradigm == paradigm)
            & (probe.dataset == dataset)
            & (probe.seed == seed)
            & (probe.task == task)
        ].iloc[0]
        seed_sd = probe[
            (probe.paradigm == paradigm) & (probe.dataset == dataset) & (probe.task == task)
        ]
        metric_names = ("auprc", "auroc") if task in CLASSIFICATION_TASKS else ("mae", "mse")
        row = {k: result[k] for k in ("paradigm", "dataset", "seed", "task", "epochs_run")}
        row["frozen_epoch"] = float(frozen["epoch"])
        for name in metric_names:
            row[f"rerun_test_{name}"] = result[f"test/{name}"]
            row[f"frozen_test_{name}"] = float(frozen[f"test/{name}"])
            row[f"diff_test_{name}"] = result[f"test/{name}"] - float(frozen[f"test/{name}"])
            row[f"frozen_seed_sd_{name}"] = float(seed_sd[f"test/{name}"].std(ddof=1))
        rows.append(row)
        print(f"[capped] {row}", flush=True)
        _write(out_dir, "capped_probe_reproduction", pd.DataFrame(rows))
    frame = pd.DataFrame(rows)
    summary: dict[str, Any] = {"n_cells": int(len(frame))}
    for name in ("auprc", "auroc", "mae", "mse"):
        column = f"diff_test_{name}"
        if column in frame:
            values = frame[column].dropna()
            summary[f"{name}_max_abs_diff"] = float(values.abs().max())
            summary[f"{name}_mean_abs_diff"] = float(values.abs().mean())
    print(f"[capped] {summary}", flush=True)
    return summary


SENSITIVITY_CELLS = (
    ("jepa", "miiv", 42, "mortality_24h"),
    ("mae", "miiv", 42, "mortality_24h"),
    ("mae", "miiv", 42, "los_remaining"),
)
SENSITIVITY_PROBE_SEEDS = (0, 1, 2, 3)


def check_capped_seed_sensitivity(args: argparse.Namespace) -> dict[str, Any]:
    """Spread of the capped probe over head-init/batch-order seeds on fixed embeddings."""
    per_seed = pd.read_parquet(FROZEN_DIR / "per_seed_results.parquet")
    probe = per_seed[
        (per_seed.experiment_class == "core_ssl_benchmark")
        & (per_seed.protocol == "linear_probe")
        & (per_seed.label_fraction == 1.0)
    ]
    rows = []
    out_dir = REPO_ROOT / args.output_dir
    for paradigm, dataset, seed, task in SENSITIVITY_CELLS:
        frozen = probe[
            (probe.paradigm == paradigm)
            & (probe.dataset == dataset)
            & (probe.seed == seed)
            & (probe.task == task)
        ].iloc[0]
        metric = "auprc" if task in CLASSIFICATION_TASKS else "mae"
        for probe_seed in SENSITIVITY_PROBE_SEEDS:
            result = run_capped_probe(args, paradigm, dataset, seed, task, probe_seed=probe_seed)
            rows.append(
                {
                    "paradigm": paradigm,
                    "dataset": dataset,
                    "seed": seed,
                    "task": task,
                    "probe_seed": probe_seed,
                    "metric": metric,
                    "epochs_run": result["epochs_run"],
                    "best_epoch": result["best_epoch"],
                    "rerun_test": result[f"test/{metric}"],
                    "frozen_test": float(frozen[f"test/{metric}"]),
                    "frozen_epoch": float(frozen["epoch"]),
                }
            )
            print(f"[capped_seed_sensitivity] {rows[-1]}", flush=True)
            _write(out_dir, "capped_probe_seed_sensitivity", pd.DataFrame(rows))
    frame = pd.DataFrame(rows)
    grouped = frame.groupby(["paradigm", "dataset", "seed", "task"])
    spread = grouped.rerun_test.agg(["min", "max", "std"])
    frozen_values = grouped.frozen_test.first()
    summary = {
        "n_runs": int(len(frame)),
        "max_range_across_probe_seeds": float((spread["max"] - spread["min"]).max()),
        "n_cells_frozen_within_probe_seed_range": int(
            ((frozen_values >= spread["min"]) & (frozen_values <= spread["max"])).sum()
        ),
        "n_cells": int(len(spread)),
    }
    print(f"[capped_seed_sensitivity] {summary}", flush=True)
    return summary


def main(argv: Sequence[str] | None = None) -> int:
    args = parse_args(argv)
    args.encoder_archive = args.encoder_archive.resolve()
    args.processed_root = args.processed_root.resolve()
    runners = {
        "encoders": check_encoders,
        "splits": check_splits,
        "embeddings": check_embeddings,
        "capped": check_capped,
        "capped_seed_sensitivity": check_capped_seed_sensitivity,
    }
    summary = {}
    out_dir = REPO_ROOT / args.output_dir
    summary_path = out_dir / "validity_summary.json"
    if summary_path.exists():
        summary = json.loads(summary_path.read_text())
    for name in args.checks:
        summary[name] = runners[name](args)
        _write(out_dir, "validity_summary", summary)
    return 0

    raise SystemExit(main(sys.argv[1:]))
